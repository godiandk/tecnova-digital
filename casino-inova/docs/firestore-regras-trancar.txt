// Regras do Cloud Firestore do projeto inova-casino.
//
// Nosso app NÃO usa o Firestore: jogador, ficha, partida e ranking estão todos
// no PostgreSQL do nosso servidor. O Firestore só foi criado junto com o projeto.
//
// Um Firestore em "modo de teste" fica aberto pra qualquer pessoa ler e escrever
// por 30 dias. A chave pública do Firebase está dentro do aplicativo, então o
// endereço não é segredo pra ninguém.
//
// Negar tudo não quebra nada aqui, justamente porque nada no app usa o Firestore.

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // Nada entra, nada sai.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
